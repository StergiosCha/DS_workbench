   [+COMPARISON]
   ?[+COMPARISON-ARTICLE]
   ¬<\/0>Ex.x
   ¬<\/1>Ex.x
THEN delete(?[+COMPARISON-ARTICLE])
   make(\/0)
   go(\/0)
   put(Ty(classical:e))
   freshput(T,entity)
   put(!)
   go(/\0)
   make(\/1)
   go(\/1)
   put(?Ty(classical:arrow(e,cn)))
ELSE abort

   ?Ty(classical:t)
   ¬<\/0>Ex.x
   ¬<\/1>Ex.x
   ¬?[+COMP]

   go(\/1)

   put(?[+BARE-VERB])

   make(\/0)
   go(\/0)
   put(?Ty(classical:e))
ELSE abort
