*clitic(REF,PERSON)
IF ?Ty(mltt:Prop)
   Tn(a)
   <\/1+>Ty(x)
   ¬?[+CLUSTER]
THEN make(\/P)
   go(\/P)
   put(?Ex.Tn(x))
   put(</\0></\1+>Tn(a))
   put(Ty(mltt:object))
   put(Fo(U_REF))
   put(?Ex.Fo(x))
   put(Person(PERSON))
   gofirst(?Ty(mltt:Prop))
ELSE abort

// A single cluster entry predicts its required second surface component.
// The first component builds both argument decorations; the second only
// discharges the lexical continuation. Completion is impossible in between.
*pontic-cluster(REF1,PERSON1,REF2,PERSON2)
IF ?Ty(mltt:Prop)
   Tn(a)
   <\/1+>Ty(x)
   ¬?[+CLUSTER]
   <\/1\/1>?Ty(mltt:arrow(object,arrow(object,Prop)))
THEN go(\/1\/1)
   make(\/0)
   go(\/0)
   IF ¬Ex.Fo(x)
   THEN put(Ty(mltt:object))
      delete(?Ty(mltt:object))
      put(Fo(U_REF1))
      put(?Ex.Fo(x))
      put(Person(PERSON1))
      put(Case(gen))
      gofirst(?Ty(mltt:Prop))
      make(\/P)
      go(\/P)
      put(?Ex.Tn(x))
      put(</\0></\1+>Tn(a))
      put(Ty(mltt:object))
      put(Fo(U_REF2))
      put(?Ex.Fo(x))
      put(Person(PERSON2))
      put(Case(acc))
      gofirst(?Ty(mltt:Prop))
      put(?[+CLUSTER])
      put(?[+CLUSTER-REF2])
   ELSE abort
ELSE abort

*cluster-tail(REF)
IF ?[+CLUSTER]
   ?[+CLUSTER-REF]
THEN delete(?[+CLUSTER-REF])
   delete(?[+CLUSTER])
ELSE abort
