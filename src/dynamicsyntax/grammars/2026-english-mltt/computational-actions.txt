intro-pred
IF ?Ty(mltt:Prop)
   ¬?[+COMP]
   ¬<\/0>Ex.x
   ¬<\/1>Ex.x
THEN make(\/1)
   go(\/1)
   put(?Ty(mltt:arrow(object,Prop)))
   go(/\1)
   make(\/0)
   go(\/0)
   put(?Ty(mltt:object))
ELSE abort

// Direct subject/object gaps in restrictive relatives. LINK confines the copy.
merge-relative
!completion
IF ?Ty(mltt:object)
   ¬Ex.Fo(x)
   (</\0>[+REL] || </\0></\1>[+REL])
   <Y>(?Ex.Tn(x) & [+REL-GAP] & Ty(x))
   subsumes(Y)
THEN merge(Y)
ELSE abort
