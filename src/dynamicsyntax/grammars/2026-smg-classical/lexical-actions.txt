*clitic(REF,PERSON,CASE)
IF ?Ty(classical:t)
   ([\/1+]?Ty(x) || Mood(Imp) || Mood(Ger))
THEN make(\/1)
   go(\/1)
   put(?Ty(classical:arrow(e,t)))
   make(\/0)
   go(\/0)
   IF ¬Ex.Fo(x)
   THEN put(Ty(classical:e))
      put(Fo(U_REF))
      put(?Ex.Fo(x))
      put(Case(CASE))
      put(Person(PERSON))
      delete(?Ty(classical:e))
      gofirst(?Ty(classical:t))
   ELSE abort
ELSE abort
